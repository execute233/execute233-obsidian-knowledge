# 线性表

线性表一般需要包含以下功能：

- **初始化线性表**：将一个线性表进行初始化，得到一个全新的线性表。
- **获取指定位置上的元素**：直接获取线性表指定位置 $i$ 上的元素。
- **获取元素的位置**：获取某个元素在线性表上的位置 $i$。
- **插入元素**：在指定位置 $i$ 上插入一个元素。
- **删除元素**：删除指定位置 $i$ 上的一个元素。
- **获取长度**：返回线性表的长度。

## 顺序表

基于数组，对其进行强化，可以为其编写一些额外的操作来强化为线性表，像这样底层依然采用顺序存储实现的线性表，称为顺序表。

以存储 `int` 的顺序表为例：

```c
typedef int E;
struct List {
    E * array;       // 实现顺序表的底层数据
    int capacity;    // 底层数组的容量
    int size;        // 存储的元素个数
};
// 使用 typedef，简化代码量
typedef struct List * ArrayList;
// 初始化操作，返回 true 即为初始化成功
_Bool initList(ArrayList list, int capacity) {
    if (capacity < 1) return 0;
    list->array = malloc(sizeof(E) * capacity); // 动态申请内存，防止局部变量销毁
    if (list->array == NULL) return 0;           // 可能会申请失败
    list->capacity = capacity;
    list->size = 0;
    return 1;
}
// 插入，注意顺序表按位序计算，即是以 1 开头
_Bool insertList(ArrayList list, E e, int order) {
    if (order > list->size + 1 || order < 1) return 0;
    if (list->size == list->capacity) {
        // 扩容操作
        const int newCapacity = list->capacity + (list->capacity >> 2);
        E * newArray = realloc(list->array, sizeof(E) * newCapacity);
        if (newArray == NULL) return 0;
        list->array = newArray;
        list->capacity = newCapacity;
    }
    // 将后面的元素向后移操作
    for (int i = list->size; i >= order; i--) {
        list->array[i] = list->array[i - 1];
    }
    list->array[order - 1] = e;
    list->size++;
    return 1;
}
// 删除指定位序的元素
void deleteList(ArrayList list, int order) {
    if (order < 1 || order > list->size) return;
    for (int i = order - 1; i < list->size - 1; i++) {
        list->array[i] = list->array[i + 1];
    }
    list->size--;
}
// 返回维护的 List 长度，NULL 时返回 -1
int sizeList(ArrayList list) {
    return list->size;
}
// 返回指定位序的元素
E * getList(ArrayList list, int order) {
    if (order < 1 || order > list->size) return NULL;
    return &list->array[order - 1];
}
// 查找某元素的位序，不存在或顺序表为 NULL 时返回 -1
int findList(ArrayList list, E e) {
    for (int i = 0; i < list->size; i++) {
        if (list->array[i] == e) return i + 1;
    }
    return -1;
}
// 清空所有元素
void clearList(ArrayList list) {
    list->size = 0;
}
// 销毁接口
void destroyList(ArrayList list) {
    if (list) {
        free(list->array);
        list->array = NULL;
        list->capacity = list->size = 0;
    }
}
```

## 链表

链表不同于顺序表，顺序表底层采用数组作为存储容器，需要分配一块连续且完整的内存空间进行使用，而链表则不需要，它通过一个指针来连接各个分散的结点，形成了一个链状的结构，每个结点存放一个元素，以及一个指向下一个结点的指针，通过这样一个一个相连，最后形成了链表。它不要申请连续的空间，只需按照顺序连接即可，虽然物理上可能不相邻，但是在逻辑上依然是每个元素相邻存放的，这样的结构叫做链表（单链表）。

链表分为带头结点的链表（会指向后续的整个链表，但头结点不存放元素）和不带头结点的链表，以带头结点的为例：

```c
typedef int E;
// 节点定义，首节点是空的
struct LinkedNode {
    E element;
    struct LinkedNode * next;
};
typedef struct LinkedNode * Node;
// 插入元素
_Bool insertList(Node head, E element, int order) {
    if (order < 1) return 0;
    // 这个循环最后会让 head 成为指定位序前面的节点
    while (--order) {
        head = head->next;
        if (head == NULL) return 0;
    }
    // 开始插入操作
    Node node = malloc(sizeof(struct LinkedNode));
    if (node == NULL) return 0;
    node->element = element;
    node->next = head->next;
    head->next = node;
    return 1;
}
// 删除指定位序的元素
_Bool deleteList(Node head, int order) {
    if (order < 1) return 0;
    // 这个循环最后会让 head 成为指定位序前面的节点
    while (--order) {
        head = head->next;
        if (head == NULL) return 0;
    }
    // 指定位序的节点是否为空
    if (head->next == NULL) return 0;
    Node preDeleteNode = head->next;
    head->next = head->next->next;
    free(preDeleteNode);
    return 1;
}
// 获取指定位序的元素
E * getList(Node head, int order) {
    if (order < 1) return NULL;
    // 这个循环会获取指定位序的节点
    do {
        head = head->next;
        if (head == NULL) return NULL;
    } while (--order);
    return &head->element;
}
// 返回存储指定元素的节点的位序
int findList(Node head, E element) {
    head = head->next;
    int i = 1;
    while (head) {
        if (head->element == element) return i;
        head = head->next;
        i++;
    }
    return -1;
}
// 给出首节点后的元素个数（要传入空首节点）
int sizeList(Node head) {
    int size = 0;
    while (head->next) {
        head = head->next;
        size++;
    }
    return size;
}
void initList(Node node) {
    node->next = NULL;
}
```

对于这两种线性表来说，频繁修改使用链表，频繁使用改查使用数组。

## 双向链表

对于链表有一个问题就是，如果我们想要操作某一个结点，比如删除或是插入，那么由于单链表的性质，我们只能先去找到它的前驱结点，才能进行。

为了解决这种查找前驱结点非常麻烦的问题，我们可以让结点不仅保存指向后结点的指针，同时也保存指向前驱结点的指针。

对于初始化的头结点，前后都为空：

```c
typedef int E;
struct ListNode {
    E element;
    struct ListNode * next;
    struct ListNode * prev;
};
typedef struct ListNode * Node;
void initList(Node head) {
    head->next = head->prev = NULL;
}
// 插入指定元素在指定位次
_Bool insertList(Node head, E element, int index) {
    if (index < 1) return 0;
    // 找到指定位序前的节点
    while (--index) {
        head = head->next;
        if (head == NULL) return 0;
    }
    // 开始关系操作
    Node nextNode = head->next;
    Node node = malloc(sizeof(struct ListNode));
    if (node == NULL) return 0;
    node->element = element;
    // 链接前面的当前节点
    node->prev = head;
    head->next = node;
    // 只有当前节点不是最后一个的时候才能与后面链接
    if (nextNode != NULL) {
        nextNode->prev = node;
        node->next = nextNode;
    } else {
        // 注意最后一个节点的 next 指针会乱指
        node->next = NULL;
    }
    return 1;
}
// 删除指定位次的元素
_Bool deleteList(Node head, int index) {
    if (index < 1) return 0;
    // 获取指定节点的前面一个节点
    while (--index) {
        head = head->next;
        if (head == NULL) return 0;
    }
    // 是否超出链表范围
    if (head->next == NULL) return 0;
    // 指定的节点
    Node preDeleteNode = head->next;
    // 判断要删除的节点是否是最后一个节点
    if (preDeleteNode->next) {
        // 不是最后一个节点
        head->next = preDeleteNode->next;
        preDeleteNode->prev = head;
    } else {
        // 是最后一个节点
        head->next = NULL;
    }
    free(preDeleteNode);
    return 1;
}
```

其他操作同单链表差不多，不再赘述。

## 循环链表

和前面的链表是一样的，但是它的最后一个结点是与头结点相连的，双向链表和单向链表都可以做成这样的环形结构。

## 栈

一种特殊的线性表，它只能在表尾（Top）进行插入（push）和删除（pop）操作，就像下面这样：

队列只能在表尾插入元素、在表头删除元素(箭头指向只能从一端进行):
- 表头:删除操作只能在这一端进行
- 表尾:插入操作只能在这一端进行

即每次插入只能在表尾部插入,删除只能在表头部删除,无法直接操作前面的元素

底部称为栈底，顶部称为栈顶，所有的操作只能在栈顶进行，也就是说，被压在下方的元素，只能等待其上方的元素出栈之后才能取出，就像我们往箱子里面放的书一样，因为只有一个口取出里面的物品，所以被压在下面的书只能等上面的书被拿出来之后才能取出，这就是栈的思想，它是一种先进后出的数据结构（FILO, First In, Last Out）。

### 顺序表实现的栈

```c
typedef int E;
struct Stack {
    E * array;
    int capacity;
    int top; // 栈顶元素下标，+1 就是大小
};
typedef struct Stack * ArrayStack;
_Bool initStack(ArrayStack stack) {
    stack->array = malloc(sizeof(E) * 10);
    if (stack->array == NULL) return 0;
    stack->capacity = 10;
    stack->top = -1;
    return 1;
}
_Bool pushStack(ArrayStack stack, E element) {
    if (stack->top + 1 == stack->capacity) {
        // 扩容
        int newCapacity = stack->capacity + (stack->capacity >> 2);
        E * newArray = realloc(stack->array, newCapacity * sizeof(E));
        if (newArray == NULL) return 0;
        stack->array = newArray;
        stack->capacity = newCapacity;
    }
    stack->array[++stack->top] = element;
    return 1;
}
E popStack(ArrayStack stack) {
    return stack->array[stack->top--];
}
void destroyStack(ArrayStack stack) {
    free(stack->array);
    stack->capacity = 0;
    stack->top = -1;
    stack->array = NULL;
}
_Bool isEmpty(ArrayStack stack) {
    return stack->top == -1;
}
```

使用数组实现的栈以外，还可以固定大小的栈，当栈已满时，就无法进行入栈操作了。

### 链表实现的栈

使用链表实现的栈（包含头结点），栈顶在前，栈底在后，示例：

```c
typedef int E;
struct ListNode {
    E element;
    struct ListNode * next;
};
typedef struct ListNode * Node;
void initStack(Node head) {
    head->next = NULL;
}
_Bool pushStack(Node head, E element) {
    struct ListNode * newNode = malloc(sizeof(struct ListNode));
    if (newNode == NULL) return 0;
    // 把节点插入到头结点后面
    newNode->next = head->next; // 无论 head->next 是否为 NULL，这行代码都适用
    head->next = newNode;
    newNode->element = element;
    return 1;
}
E popStack(Node head) {
    Node tmp = head->next;
    if (tmp == NULL) return 0; // 注意栈为空的情况
    // 插入保证了链表末尾节点指向 NULL，可以直接赋值
    head->next = tmp->next;
    E element = tmp->element;
    free(tmp);
    return element;
}
_Bool isEmpty(Node head) {
    return head->next == NULL;
}
void destroyStack(Node head) {
    Node cleanerPtr;
    do {
        cleanerPtr = head->next;
        free(head);
        head = cleanerPtr;
    } while (cleanerPtr != NULL);
}
```

## 队列

跟排队一样，先进先出（FIFO, First In, First Out），同样有 push 和 pop。

对于使用顺序表实现的队列，经过多次出队入队操作，指针会指向数组的最后，延长数组不是个办法，所以一般采用循环队列：

![Exported image](_assets/%E7%BA%BF%E6%80%A7%E8%A1%A8/%E7%BA%BF%E6%80%A7%E8%A1%A8__11-43-25-3.png)

可以在移动队首队尾指针时，考虑循环的问题，也就是说如果到达了数组尽头，那么就直接从数组的前面重新开始计算，这样就相当于逻辑上都循环了，队首和队尾指针在一开始的时候都指向同一个位置，每入队一个新的元素，依然是先让队尾后移一位，在所指向位置插入元素，出队同理。

不过这样仍有问题，如何判断队列已满了？

- 牺牲一个存储单元，如果队尾指针在队首指针后面，就是满了
- 用一个计数器，达到最大容量则满了

### 数组实现的队列

```c
typedef int E;
struct Queue {
    E * array;
    int capacity;  // 数组容量
    int front, rear; // 队首指针，队尾指针
};
typedef struct Queue * ArrayQueue;

_Bool initQueue(ArrayQueue queue, int capacity) {
    // 这里容量要 +1，因为需要牺牲一个存储空间来记录是否满了
    capacity++;
    queue->array = malloc(sizeof(E) * capacity);
    if (queue->array == NULL) return 0;
    queue->capacity = capacity;
    queue->front = queue->rear = 0;
    return 1;
}
_Bool offerQueue(ArrayQueue queue, E element) {
    // 先计算队尾指针向后移动，考虑索引出界情况
    int pos = (queue->rear + 1) % queue->capacity;
    // 判断是否已满，指针不能重合，要不然就是初始尾空的情况了
    if (pos == queue->front) return 0;
    // 入队操作，注意队尾指针是不包含元素的，要在队尾前面给元素
    queue->array[queue->rear] = element;
    queue->rear = pos;
    return 1;
}
E pollQueue(ArrayQueue queue) {
    // 如果队列为空就返回 0
    if (queue->rear == queue->front) return 0;
    E result = queue->array[queue->front];
    // 计算队头指针
    queue->front = (queue->front + 1) % queue->capacity;
    return result;
}
```

### 链表实现的队列

```c
typedef int E;
struct LNode {
    E element;
    struct LNode * next;
};
typedef struct LNode * Node;
struct Queue {
    Node front, rear;
};
typedef struct Queue * LinkedQueue;

_Bool initQueue(LinkedQueue queue) {
    Node node = malloc(sizeof(struct LNode));
    if (node == NULL) return 0;
    node->next = NULL;
    queue->front = queue->rear = node;
    return 1;
}
_Bool offerQueue(LinkedQueue queue, E element) {
    Node node = malloc(sizeof(struct LNode));
    if (node == NULL) return 0;
    node->next = NULL;
    node->element = element;

    queue->rear->next = node;
    queue->rear = node;
    return 1;
}
E pollQueue(LinkedQueue queue) {
    // 获得队头元素节点
    Node tmp = queue->front->next;
    if (tmp == NULL) return 0; // soc，没东西该咋办？
    E element = tmp->element;

    queue->front->next = tmp->next;
    // 如果队尾就是要跑出去的节点
    if (queue->rear == tmp) queue->rear = queue->front;
    free(tmp);
    return element;
}
_Bool isEmpty(LinkedQueue queue) {
    return queue->front == queue->rear;
}
```
