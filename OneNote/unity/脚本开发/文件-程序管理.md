# 文件-程序管理

// 游戏数据文件夹路径，就是项目里的Assets文件夹，打包后就是加密压缩只读的
Debug._Log_(Application.dataPath);
// 持久化文件夹路径，系统分配给应用的临时文件夹
Debug._Log_(Application.persistentDataPath);
// StreamingAssets，只读非加密的文件夹路径，一般在Assets/StreamingAsset
Debug._Log_(Application.streamingAssetsPath);
// 临时文件夹
Debug._Log_(Application.temporaryCachePath);
// 是否在后台运行
Debug._Log_(Application.runInBackground);
// 浏览器打开url
Application._OpenURL_("https://www.google.com/");
// 退出游戏
Application._Quit_();
