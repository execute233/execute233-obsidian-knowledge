# 按键监听

```cs
// 在每一帧监听键盘、鼠标与虚拟按键
void Update()
{
    // ===== 鼠标 =====
    // 0 = 左键,1 = 右键,2 = 中键滚轮
    if (Input.GetMouseButtonDown(0))
    {
        Debug.Log("left click"); // 按下瞬间触发
    }
    if (Input.GetMouseButton(0))
    {
        Debug.Log("left click pressing"); // 持续按下
    }
    if (Input.GetMouseButtonUp(0))
    {
        Debug.Log("left up click"); // 抬起时触发
    }

    // ===== 键盘 =====
    if (Input.GetKeyDown(KeyCode.A))
    {
        Debug.Log("a click"); // 按下瞬间触发
    }
    if (Input.GetKey(KeyCode.A))
    {
        Debug.Log("a click pressing"); // 持续按下
    }
    if (Input.GetKeyUp(KeyCode.A))
    {
        Debug.Log("a up click"); // 抬起时触发
    }

    // ===== 轴(一般在 [-1, 1] 之间变动)=====
    float horizontal = Input.GetAxis("Horizontal"); // 水平
    float vertical = Input.GetAxis("Vertical");     // 垂直

    // ===== 虚拟按键(在 Input Manager 里配置)=====
    if (Input.GetButtonDown("Jump"))
    {
        Debug.Log("jump");
    }
}
```
