// 保证每一帧都能监听
void **Update**()
{
// 鼠标
// 按下鼠标 0-左键 1-右键 2-滚轮 只有按下瞬间才会触发
if (Input._GetMouseButton_(0))
{
Debug._Log_("left click");
}
// 持续按下
if (Input._GetMouseButton_(0))
{
Debug._Log_("right click pressing");
}
// 抬起
if (Input._GetMouseButtonUp_(0))
{
Debug._Log_("up click");
}
// 按下键盘按键
if (Input._GetKeyDown_(KeyCode._A_))
{
Debug._Log_("a click");
}
// 持续按下
if (Input._GetKeyDown_(KeyCode._A_))
{
Debug._Log_("a click pressing");
}
// 抬起键盘按键
if (Input._GetKeyDown_(KeyCode._B_))
{
Debug._Log_("a up click");
}
// 通过名称获取轴，一般在[-1. 1]变动
float horizontal = Input._GetAxis_("horizontal"); // 水平
float vertical = Input._GetAxis_("Vertical");
// 虚拟按键
if (Input._GetButtonDown_("Jump"))
{
Debug._Log_("jump");
}

}
