// 自定义物体，在面板中可以自定义
public GameObject **Cube**;
// 自定义预设体
public GameObject **Prefab**;
void **Start**()
{
// 代表使用该脚本的物体
GameObject o = gameObject;
Debug._Log_(o.name);
Debug._Log_(o.tag);
Debug._Log_(o.layer);   Debug._Log_(Cube.name);
// 真正激活状态
Debug._Log_(Cube.activeInHierarchy);
// 自身的激活状态
Debug._Log_(Cube.activeSelf);
// 获取transform组件
Debug._Log_(Cube.transform.position);
// 获取其他组件
GetComponent<BoxCollider>();
// 获取子物体身上的某个组件
GetComponentInChildren<CapsuleCollider>();
// 获取父物体身上的某个组件
GetComponentInParent<BoxCollider>();
// 添加组件
gameObject.AddComponent<AudioSource>();
// 通过名字获取物体
GameObject._Find_("Target");
// 通过tag获取物体
GameObject._FindGameObjectsWithTag_("Target");
// 通过预设体实例化游戏物体
GameObject newGameObject = _Instantiate_(Prefab);
// 销毁物体
_Destroy_(newGameObject);
}
