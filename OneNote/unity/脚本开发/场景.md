# 场景

场景加载/卸载一般在 [[unity/脚本开发/初步认识|Start]] 和 [[unity/脚本开发/初步认识|Update]] 中处理,场景切换时 transform 父子关系会失效,见 [[unity/脚本开发/transform控制父子|transform控制父子]]。

```cs
using UnityEngine.SceneManagement;

public class ScenesTest : MonoBehaviour
{
    AsyncOperation operation;

    void Start()
    {
        // 同步加载场景:可指定索引或场景名,直接跳转到该场景
        // LoadSceneMode.Single = 替换当前场景(默认);Additive = 叠加到当前场景
        SceneManager.LoadScene(0);

        // 获取当前场景
        Scene scene = SceneManager.GetActiveScene();
        Debug.Log(scene.name);       // 场景名称
        Debug.Log(scene.isLoaded);   // 是否已加载
        Debug.Log(scene.path);       // 场景路径(项目里的相对路径)
        Debug.Log(scene.buildIndex); // 场景索引
        // 获取该场景所有根物体
        scene.GetRootGameObjects();

        // 场景管理类
        Debug.Log(SceneManager.sceneCount); // 已加载场景数

        // 创建新场景
        SceneManager.CreateScene("newScene");
        // 卸载场景
        SceneManager.UnloadSceneAsync("newScene");

        // 异步加载
        StartCoroutine(loadScene());
    }

    // 协程方法用于异步加载场景,必须返回 IEnumerator
    IEnumerator loadScene()
    {
        operation = SceneManager.LoadSceneAsync(1);
        // 加载完后不立即跳转,等待 allowSceneActivation 置 true
        operation.allowSceneActivation = false;
        yield return operation;
    }

    // Update is called once per frame
    void Update()
    {
        // 每帧获取加载进度(0 到 0.9 表示加载中,0.9 表示加载完成等待跳转)
        Debug.Log(operation.progress);
        // 在指定条件下激活跳转
        if (operation.progress >= 0.9f)
        {
            operation.allowSceneActivation = true;
        }
    }
}
```
