# 场景

```cs

using UnityEngine.SceneManagement;
public class ScenesTest : MonoBehaviour
{
AsyncOperation operation;
void Start()
{
// 同步加载场景,指定索引或者场景名字,直接跳转到这个场景里
// 可以指定加载方式,默认LoadSceneMode.Single,Additive是把加载的场景添加到当前场景里
SceneManager._LoadScene_(0);
// 获取当前场景
Scene scene = SceneManager._GetActiveScene_();
// 场景名称
Debug._Log_(scene.name);
// 场景是否被加载
Debug._Log_(scene.isLoaded);
// 场景路径,就是项目里
Debug._Log_(scene.path);
// 场景索引
Debug._Log_(scene.buildIndex);
// 获取该场景说有物体
scene.GetRootGameObjects();
// 场景管理类
// 已加载场景数
Debug._Log_(SceneManager.sceneCount);
// 创建新场景
SceneManager._CreateScene_("newScene");
// 卸载场景
SceneManager._UnloadSceneAsync_("newScene");
// 异步加载
StartCoroutine(loadScene());
}
// 协程方法来异步加载场景,必须返回IEnumerator
IEnumerator loadScene()
{
operation = SceneManager._LoadSceneAsync_(1);
// 加载完场景后不要进行跳转
operation.allowSceneActivation = false;
yield return operation;
}

// Update is called once per frame
void Update()
{
// 可以每一帧获取加载进度
Debug._Log_(operation.progress);
// 在指定条件下加载场景
if (operation.progress >= 0.9f)
{
operation.allowSceneActivation = true;
}   }
}
```
