public void initBinder(WebDataBinder binder) {
// 添加自定义转换器

}

static class MyDateFormatter implements Formatter\<LocalDateTime\> {

private static final DateTimeFormatter _formatter_ = DateTimeFormatter._ofPattern_("yyyy|MM|dd");
@Override
public LocalDateTime parse(String text, Locale locale) throws ParseException {
return LocalDateTime._parse_(text, _formatter_);
}
@Override
public String print(LocalDateTime object, Locale locale) {
return _formatter_.format(object);
}
}
